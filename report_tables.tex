\begin{table}[htbp]

\caption{Operational risk tiers in the Decision Support System
(probability $= P(\mathrm{Continuation}=1)$).}
\label{tab:risk_tiers}
\begin{tabular}{lp{8.5cm}}
\toprule
\textbf{Probability range} & \textbf{Risk tier} \\
\midrule
$p \ge 0.73$ & Low Risk (high likelihood to continue pathway) \\
$0.40 \le p < 0.73$ & Medium Risk (transitional / uncertain) \\
$p < 0.40$ & High Risk (likely to drop out / reverse decision) \\
\bottomrule
\end{tabular}
